Koneksi Database Supabase

Sinkronisasi real-time antar HP dan Komputer

Skema SQL Tabel Supabase (Jalankan di SQL Editor) 
-- Tabel Pengaturan Harga Produk
CREATE TABLE IF NOT EXISTS marjayuk_settings (
  id INT PRIMARY KEY DEFAULT 1,
  price_small NUMERIC NOT NULL DEFAULT 5000,
  price_large NUMERIC NOT NULL DEFAULT 10000,
  updated_at TIMESTAMPTZ DEFAULT NOW()
);
INSERT INTO marjayuk_settings (id, price_small, price_large) VALUES (1, 5000, 10000) ON CONFLICT (id) DO NOTHING;

-- Tabel Pesanan & Transaksi Pelanggan
CREATE TABLE IF NOT EXISTS marjayuk_orders (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  customer_name TEXT NOT NULL,
  customer_identity TEXT,
  order_date DATE NOT NULL DEFAULT CURRENT_DATE,
  qty_small INT NOT NULL DEFAULT 0,
  qty_large INT NOT NULL DEFAULT 0,
  unit_price_small NUMERIC NOT NULL DEFAULT 5000,
  unit_price_large NUMERIC NOT NULL DEFAULT 10000,
  total_items INT NOT NULL DEFAULT 0,
  total_price NUMERIC NOT NULL DEFAULT 0,
  payment_status TEXT NOT NULL DEFAULT 'Belum Dibayar',
  order_status TEXT NOT NULL DEFAULT 'Menunggu',
  notes TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Tabel Riwayat Perubahan Data (Audit Log)
CREATE TABLE IF NOT EXISTS marjayuk_audit_logs (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  action_type TEXT NOT NULL,
  description TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Aktifkan Realtime & Akses Publik (Atau sesuaikan RLS Auth Anda)
ALTER TABLE marjayuk_orders ENABLE ROW LEVEL SECURITY;
ALTER TABLE marjayuk_settings ENABLE ROW LEVEL SECURITY;
ALTER TABLE marjayuk_audit_logs ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Akses Penuh Pesanan" ON marjayuk_orders FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Akses Penuh Pengaturan" ON marjayuk_settings FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Akses Penuh Audit" ON marjayuk_audit_logs FOR ALL USING (true) WITH CHECK (true);
